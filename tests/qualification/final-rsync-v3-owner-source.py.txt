"""Direct inherited-rsync original controls under standing Unison8 + EXglobal."""
import argparse,hashlib,json,os,sys,time
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
sys.path[:0]=[str(ROOT/'tests'),str(ROOT/'tests/sdk_host_matrix')]
from final_profile_owner import sha,closed
from cached_benchmark.contract import read_bytes
from shared_cpu_pool import Pool,authorized_unison_tokens
from shared_io_pool import GlobalPool,CANONICAL_MANIFEST,CANONICAL_SHA256,try_io_cpu_bundle
from shared_process_owner import retain_after_record
from sdk_merge_receiving.store import Store
from shared_child import start


def main():
    p=argparse.ArgumentParser();p.add_argument('--plan',required=True);p.add_argument('--sha256',required=True);a=p.parse_args()
    raw=read_bytes(a.plan,1048576)
    if hashlib.sha256(raw).hexdigest()!=a.sha256:raise ValueError('exact plan SHA required')
    packet=json.loads(raw)
    if packet['repository']!=str(ROOT) or packet['state']!='PREPARED_V3_AFTER_ACTUAL_PHOTO_HANDOVER':raise ValueError('exact handover plan required')
    def sources():
        for name,digest in packet['source_pins'].items():
            if sha(ROOT/name)!=digest:raise ValueError('source changed: '+name)
    sources()
    policy=ROOT.parent/'photo/preview-generator/SHARED-DEVELOPMENT-RESOURCE-POLICY-v3.json'
    if sha(policy)!='1dbe12eb97dfe885a0ec94057c476f4b825adc2d37085f0ecd48fbdf7b047352':raise ValueError('policy changed')
    if tuple(authorized_unison_tokens(read_bytes(str(policy),1048576)))!=tuple(range(8)):raise ValueError('standing8 differs')
    pool=Pool.read(str(ROOT.parent/'photo/preview-generator/engine-v3-native-control/ROOT-CPU-POOL-v1.json'),'4d19610edf6b400f99c00962cf52644ad93c924786b376faba91577808b693fd')
    io_pool=GlobalPool.read(CANONICAL_MANIFEST,CANONICAL_SHA256)
    root=Path(packet['owner_workspace']);root.mkdir(mode=0o700)
    store=io=cpu=child=None
    try:
        st=root.stat();store=Store(str(root),dict(device=st.st_dev,inode=st.st_ino))
        bundle=try_io_cpu_bundle(io_pool,pool,tuple(range(8)))
        if bundle is None:
            store.write('CONTENTION.json',dict(no_child_attempted=True));store.close();return 2
        io,cpu=bundle;sources()
        if sha(packet['interpreter']['path'],64*1048576)!=packet['interpreter']['sha256']:raise ValueError('interpreter changed')
        for pin in packet['build_receipts']:
            data=read_bytes(pin['path'],1048576)
            if len(data)!=pin['bytes'] or hashlib.sha256(data).hexdigest()!=pin['sha256']:raise ValueError('build receipt changed')
            build=json.loads(data)
            if not(build['complete'] is True and build['build_exit']==0 and build['selftest_exit']==0):raise ValueError('build qualification differs')
            if sha(build['compiler'],64*1048576)!=build['compiler_sha256']:raise ValueError('compiler changed')
            for name,digest in build['source_sha256'].items():
                if name.startswith('src/') and sha(ROOT/name)!=digest:raise ValueError('production source changed')
        deadline=time.monotonic()+packet['resources']['owner_observation_seconds']
        store.write('START.json',dict(owner_pid=os.getpid(),command=packet['command'],plan_sha256=a.sha256,
            cpu_indexes=list(range(8)),io_mode='EX_GLOBAL',deadline=deadline))
        print('START',os.getpid(),str(root),flush=True)
        child=start(packet['command'],dict(os.environ,PYTHONDONTWRITEBYTECODE='1'),pass_fds=(*cpu.fds,*io.fds))
        out=child.finish(deadline);store.write('RUNNER.json',out);closed(out,0)
        if out['command']!=packet['command']:raise ValueError('runner command changed')
        work=Path(out['stdout'].strip().splitlines()[-1])
        if str(work.parent)!='/private/tmp' or not work.name.startswith('unison-rsync-publication-'):raise ValueError('unexpected result namespace')
        result=json.loads(read_bytes(str(work/'RESULT.json'),16*1048576))
        if (result['complete'] is not True or result['phases']!=packet['phases_per_compiler']
                or len(result['calls'])!=20 or result['SDK_invoked'] is not False
                or result['transport_invoked'] is not False or result['real_media_opened'] is not False):
            raise ValueError('rsync scope incomplete')
        for path,digest in {**result['source_pins'],**result['linked_artifacts']}.items():
            if sha(path,64*1048576)!=digest:raise ValueError('linked artifact/source changed')
        for call in result['calls']:closed(call,0)
        for index,label in enumerate(('stock','ox')):
            calls=result['calls'][index*10:(index+1)*10]
            for phase,call in zip(packet['phases_per_compiler'],calls[2:]):
                if call['command'][2]!=phase or ('PASS streaming receiver rsync '+phase+'; actual ticket/channel/native ownership') not in call['stdout']:
                    raise ValueError('actual phase missing or reordered')
        sources()
        audit=dict(complete=True,nested_calls=20,runner=out['client_cleanup'],phases_per_compiler=result['phases'],
            result=dict(path=str(work/'RESULT.json'),sha256=sha(work/'RESULT.json',16*1048576)),
            linked_artifacts=len(result['linked_artifacts']),production_source_pins=len(result['source_pins']),
            unknown_native_fixture_ownership_retained_until_process_exit=True,external_transport=False,
            parent_tool_terminal_proved=False)
        store.write('RESULT.json',audit)
        store.close();cpu.close_after_verified_settlement(owner_reaped=True,groups_settled=True,external_children_settled=True)
        io.close_after_verified_settlement(owner_reaped=True,groups_settled=True,external_children_settled=True)
        print('CLOSED',str(root),str(work),flush=True);return 0
    except BaseException as error:retain_after_record(lambda:print('RETAIN',repr(error),str(root),flush=True))


if __name__=='__main__':raise SystemExit(main())
